# striprtf powers the pure-Python RTF fallback (extract_rtf.py) when pandoc is
# absent. Tiny, dependency-free, plain-text only — pandoc remains the
# preferred RTF route when on PATH. Legacy .doc has no pure-Python reader and
# relies on a system converter (soffice/textutil/antiword), so nothing is
# pinned here for it.
striprtf>=0.0.26,<1.0
# Pillow powers save_image_safe() — every extractor funnels image bytes
# through it to enforce the 2000-px max-dimension cap (Claude's vision
# encoder breaks above ~2 k px) and to re-encode palette/RGBA images that
# would otherwise be unviewable. Pillow is a transitive dep of pymupdf and
# pymupdf-layout but we pin it explicitly so the lightweight tier stays
# resilient if a future pymupdf release drops the dependency.
Pillow>=10.0,<14.0
tiktoken>=0.7.0,<1.0
# Pin lxml — transitive dep of trafilatura, historically XXE-vulnerable
